        PATH = "${JAVA_HOME}\\bin;${env.PATH}"
    }
    stages {

            steps {
                git branch: 'main', url: 'https://github.com/lakhan-porwal/ORSProject04-Complete.git'

            steps {

                    bat 'docker build -t webapp:latest .'
                }
            }

        stage('Deploy with Docker Compose') {
            steps {
                dir('ORSProject04') {
                    bat 'docker compose -p orsproject04 up -d --no-build'
                    bat 'docker compose -p orsproject04 ps -a'
                }
            }
        }
        stage('Verify Images') {
            steps { bat 'docker images' }
        }
