JOB: AWS-MySQL-Setup
NOTE: Newly created MySQL containers use host port 3308 -> container port 3306.
Existing containers are started as-is; docker start does not change their port mapping.
pipeline {
    agent any
    stages {
        stage('Setup MySQL on AWS') {
            steps {
                script {
                    def AWS_HOST = '3.87.152.16'
                    def SSH_KEY = 'C:/Users/Mohit porwal/Downloads/ORSProject04-key.pem'
                    def result = bat(returnStatus: true, script: """
                        ssh -i "${SSH_KEY}" -o StrictHostKeyChecking=no -o ConnectTimeout=15 ubuntu@${AWS_HOST} "set -e; sudo docker network inspect ors-network >/dev/null; if sudo docker container inspect mysql-container >/dev/null 2>&1; then echo MySQL container exists; sudo docker start mysql-container; else sudo docker volume create mysql-data; sudo docker run -d --name mysql-container --network ors-network --network-alias mysql -p 3308:3306 -e MYSQL_ROOT_PASSWORD=root -e MYSQL_DATABASE=p04 -v mysql-data:/var/lib/mysql --restart unless-stopped mysql:8.0; fi; sudo docker network connect ors-network mysql-container 2>/dev/null || true; sudo docker ps --filter name=mysql-container"
                    """)
                    if (result != 0) { error('MySQL setup failed. Check Console Output.') }
                }
            }
        }
        stage('Check MySQL Readiness') {
            steps {
                script {
                    def AWS_HOST = '3.87.152.16'
                    def SSH_KEY = 'C:/Users/Mohit porwal/Downloads/ORSProject04-key.pem'
                    def result = bat(returnStatus: true, script: """
                        ssh -i "${SSH_KEY}" -o StrictHostKeyChecking=no -o ConnectTimeout=15 ubuntu@${AWS_HOST} "sudo docker exec mysql-container mysqladmin ping -h 127.0.0.1 -uroot -proot --silent && sudo docker exec mysql-container mysql -uroot -proot -e 'SHOW DATABASES;'"
                    """)
                    if (result != 0) { error('MySQL readiness check failed. Check Console Output.') }
                }
            }
        }
    }
    post {
        success { echo 'SUCCESS: MySQL setup and readiness check passed.' }
        failure { echo 'FAILED: MySQL setup/readiness failed.' }
    }
}
