JOB: ORS-DockerHub-Push
REQUIREMENT: Jenkins credential ID dockerhub-creds has Docker Hub username and read/write token.
Requires local images webapp:latest and mysql:8.0.
pipeline {
    agent any
    stages {
        stage('Check Local Images') {
            steps { bat 'docker images' }
        }
        stage('Login to Docker Hub') {
            steps {
                withCredentials([usernamePassword(credentialsId: 'dockerhub-creds', usernameVariable: 'DOCKER_USER', passwordVariable: 'DOCKER_TOKEN')]) {
                    bat 'echo %DOCKER_TOKEN% | docker login -u %DOCKER_USER% --password-stdin'
                }
            }
        }
        stage('Tag Images') {
            steps {
                bat 'docker tag webapp:latest lakhanporwal/webapp:latest && docker tag mysql:8.0 lakhanporwal/mysql:latest'
            }
        }
        stage('Push Images') {
            options { timeout(time: 30, unit: 'MINUTES') }
            steps {
                retry(3) {
                    bat 'docker push lakhanporwal/webapp:latest'
                    bat 'docker push lakhanporwal/mysql:latest'
                }
            }
        }
        stage('Verify Images') {
            steps {
                bat 'docker images lakhanporwal/webapp && docker images lakhanporwal/mysql'
            }
        }
    }
    post {
        success { echo 'SUCCESS: Webapp and MySQL images pushed to Docker Hub!' }
        failure { echo 'FAILED: Check Console Output.' }
    }
}
