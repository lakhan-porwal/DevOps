JOB: AWS-ORS-Full-Deployment
Deploys Webapp and verifies MySQL. It preserves an existing MySQL container and its volume, and recreates only webapp-container.
Requires ors-network. MySQL Workbench port is 3308; Webapp uses internal MySQL port 3306.
pipeline {
    agent any
    stages {
        stage('Check AWS Docker and Network') {
            steps {
                bat 'ssh -i "C:/Users/Mohit porwal/Downloads/ORSProject04-key.pem" -o StrictHostKeyChecking=no -o ConnectTimeout=15 ubuntu@3.87.152.16 "sudo docker --version && sudo docker network inspect ors-network >/dev/null"'
            }
        }
        stage('Setup MySQL Container') {
            steps {
                bat 'ssh -i "C:/Users/Mohit porwal/Downloads/ORSProject04-key.pem" -o StrictHostKeyChecking=no ubuntu@3.87.152.16 "set -e; if sudo docker container inspect mysql-container >/dev/null 2>&1; then sudo docker start mysql-container; else sudo docker volume create mysql-data; sudo docker run -d --name mysql-container --network ors-network --network-alias mysql -p 3308:3306 -e MYSQL_ROOT_PASSWORD=root -e MYSQL_DATABASE=p04 -v mysql-data:/var/lib/mysql --restart unless-stopped mysql:8.0; fi; sudo docker network connect ors-network mysql-container 2>/dev/null || true"'
            }
        }
        stage('Check MySQL Readiness') {
            steps {
                bat 'ssh -i "C:/Users/Mohit porwal/Downloads/ORSProject04-key.pem" -o StrictHostKeyChecking=no ubuntu@3.87.152.16 "sudo docker exec mysql-container mysqladmin ping -h 127.0.0.1 -uroot -proot --silent && sudo docker exec mysql-container mysql -uroot -proot -e \'SHOW DATABASES;\'"'
            }
        }
        stage('Pull Webapp Image') {
            steps {
                bat 'ssh -i "C:/Users/Mohit porwal/Downloads/ORSProject04-key.pem" -o StrictHostKeyChecking=no ubuntu@3.87.152.16 "sudo docker pull lakhanporwal/webapp:latest"'
            }
        }
        stage('Deploy Webapp') {
            steps {
                bat 'ssh -i "C:/Users/Mohit porwal/Downloads/ORSProject04-key.pem" -o StrictHostKeyChecking=no ubuntu@3.87.152.16 "sudo docker rm -f webapp-container 2>/dev/null || true; sudo docker run -d --name webapp-container --network ors-network -p 8080:8080 --restart unless-stopped lakhanporwal/webapp:latest"'
            }
        }
        stage('Verify Deployment') {
            steps {
                bat 'ssh -i "C:/Users/Mohit porwal/Downloads/ORSProject04-key.pem" -o StrictHostKeyChecking=no ubuntu@3.87.152.16 "sudo docker ps -a; echo WEBAPP_LOGS; sudo docker logs --tail 40 webapp-container"'
            }
        }
    }
    post {
        success { echo 'SUCCESS: Deployment commands completed. Verify app and database connectivity.' }
        failure { echo 'FAILED: Check the failed stage and Console Output.' }
    }
}
